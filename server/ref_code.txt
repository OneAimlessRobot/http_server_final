#include <openssl/ssl.h>
#include <openssl/err.h>

int main() {
    // 1. Initialize OpenSSL
    SSL_library_init();
    OpenSSL_add_all_algorithms();
    SSL_load_error_strings();
    const SSL_METHOD *method = TLS_server_method();
    SSL_CTX *ctx = SSL_CTX_new(method);

    // 2. Load the certificates you created in Step 2
    SSL_CTX_use_certificate_file(ctx, "cert.pem", SSL_FILETYPE_PEM);
    SSL_CTX_use_PrivateKey_file(ctx, "key.pem", SSL_FILETYPE_PEM);

    // 3. Create a standard socket bind/listen on port 443
    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    // bind to port 443... listen...

    while(1) {
        int client_fd = accept(server_fd, ...);

        // 4. Wrap the standard socket file descriptor in an OpenSSL object
        SSL *ssl = SSL_new(ctx);
        SSL_set_fd(ssl, client_fd);

        // 5. Perform the TLS Handshake
        if (SSL_accept(ssl) <= 0) {
            ERR_print_errors_fp(stderr);
        } else {
            // 6. Use SSL_read and SSL_write instead of recv/send
            char buffer[1024];
            SSL_read(ssl, buffer, sizeof(buffer));
            
            char *response = "HTTP/1.1 200 OK\r\nContent-Length: 13\r\n\r\nHello HTTPS!";
            SSL_write(ssl, response, strlen(response));
        }

        // 7. Clean up the connection
        SSL_shutdown(ssl);
        SSL_free(ssl);
        close(client_fd);
    }
    SSL_CTX_free(ctx);
}
